namespace DuressSecurityApp

    public enum LockContext
    {
        ScreenLock,
        AppLock,
        InputOnly
    }

    public partial class PinPopupPage : ContentPage
    {
        public string ResultPin { get; private set; } = string.Empty;
        private readonly LockContext _context;
        private readonly string _targetPackageName;
        private readonly StringBuilder _pinBuffer = new();

        // 1. المشد الخالي (Default Constructor) لضمان التوافق مع XAML
        public PinPopupPage() : this("أدخل الرمز الرئيسي للتحقق", LockContext.InputOnly, "")
        {
        }

        // 2. المشد الرئيسي المنشئ مع البارامترات
        public PinPopupPage(string title, LockContext context = LockContext.InputOnly, string targetPackageName = "")
        {
            InitializeComponent();
            _context = context;
            _targetPackageName = targetPackageName;

            if (TitleLabel != null)
                TitleLabel.Text = title;

            // إخفاء زر الإلغاء تماماً عند القفل الإجباري لمنع التجاوز
            if (_context != LockContext.InputOnly && CancelButton != null)
            {
                CancelButton.IsVisible = false;
            }

            UpdatePinDisplay();
        }

        protected override void OnAppearing()
        {
            base.OnAppearing();
            // إغلاق شريط الإشعارات فور ظهور النافذة لمنع التجاوز
            CollapseStatusBar();
        }

        protected override bool OnBackButtonPressed()
        {
            // منع زر الرجوع نهائياً إذا كانت النافذة للقفل الإجباري
            if (_context != LockContext.InputOnly)
            {
                CollapseStatusBar();
                return true;
            }
            return base.OnBackButtonPressed();
        }

        private void OnNumClicked(object sender, EventArgs e)
        {
            if (sender is Button btn && !string.IsNullOrEmpty(btn.Text))
            {
                // السماح بالحروف والأرقام حتى طول 4 خانات
                if (_pinBuffer.Length < 4)
                {
                    _pinBuffer.Append(btn.Text);
                    UpdatePinDisplay();
                }
            }
        }

        private void OnDeleteClicked(object sender, EventArgs e)
        {
            if (_pinBuffer.Length > 0)
            {
                _pinBuffer.Remove(_pinBuffer.Length - 1, 1);
                UpdatePinDisplay();
            }
        }

        private async void OnCancelClicked(object sender, EventArgs e)
        {
            if (_context == LockContext.InputOnly)
            {
                _pinBuffer.Clear();
                ResultPin = string.Empty;
                UpdatePinDisplay();
                await Navigation.PopModalAsync();
            }
        }

        private async void OnConfirmClicked(object sender, EventArgs e)
        {
            if (_pinBuffer.Length > 0)
            {
                if (_context == LockContext.InputOnly)
                {
                    ResultPin = _pinBuffer.ToString();
                    await Navigation.PopModalAsync();
                }
                else
                {
                    ProcessPinInput(_pinBuffer.ToString());
                }
            }
        }

        private void UpdatePinDisplay()
        {
            if (PinDisplay != null)
            {
                // إظهار دوائر القفل بناءً على طول النص المدخل
                PinDisplay.Text = new string('•', _pinBuffer.Length);
            }
        }

        private async void ProcessPinInput(string enteredPin)
        {
            string masterPin = Preferences.Get("PIN_TYPE_0", "8888");
            string duressPin = Preferences.Get("PIN_TYPE_1", "");
            string lockPin = Preferences.Get("PIN_TYPE_2", "");
            string formatPin = Preferences.Get("PIN_TYPE_3", "");

            // 1. رمز الفورمات
            if (!string.IsNullOrEmpty(formatPin) && enteredPin == formatPin)
            {
                if (_context == LockContext.ScreenLock)
                {
                    TriggerFullDeviceWipe();
                }
                else
                {
                    WipeTargetAppData(_targetPackageName);
                }
                return;
            }

            // 2. رمز الإكراه
            if (!string.IsNullOrEmpty(duressPin) && enteredPin == duressPin)
            {
                await Navigation.PopModalAsync();
                if (Microsoft.Maui.Controls.Application.Current?.Windows.Count > 0)
                {
                    Microsoft.Maui.Controls.Application.Current.Windows[0].Page = new NavigationPage(new FakeSandboxPage());
                }
                return;
            }

            // 3. الرمز الصحيح
            if (enteredPin == masterPin || (!string.IsNullOrEmpty(lockPin) && enteredPin == lockPin))
            {
                ResultPin = enteredPin;
                await Navigation.PopModalAsync();
                return;
            }

            // 4. رمز خاطئ
            _pinBuffer.Clear();
            UpdatePinDisplay();
            try { Vibration.Default.Vibrate(TimeSpan.FromMilliseconds(300)); } catch { }
        }

        private void CollapseStatusBar()
        {
#if ANDROID
            try
            {
                var context = Android.App.Application.Context;
                var statusBarService = context.GetSystemService("statusbar");
                var statusBarManager = Java.Lang.Class.ForName("android.app.StatusBarManager");
                var collapseMethod = statusBarManager?.GetMethod("collapsePanels");
                collapseMethod?.Invoke(statusBarService);
            }
            catch { }
#endif
        }

        private void TriggerFullDeviceWipe()
        {
#if ANDROID
            try
            {
                var context = Android.App.Application.Context;
                var dpm = (DevicePolicyManager?)context.GetSystemService(Context.DevicePolicyService);
                dpm?.WipeData(WipeDataFlags.WipeExternalStorage);
            }
            catch (Exception ex)
            {
                System.Diagnostics.Debug.WriteLine($"Wipe Error: {ex.Message}");
            }
#endif
        }

        private void WipeTargetAppData(string packageName)
        {
#if ANDROID
            try
            {
                var context = Android.App.Application.Context;
                if (!string.IsNullOrEmpty(packageName))
                {
                    var am = (ActivityManager?)context.GetSystemService(Context.ActivityService);
                    am?.ClearApplicationUserData();
                }

                var homeIntent = new Intent(Intent.ActionMain);
                homeIntent.AddCategory(Intent.CategoryHome);
                homeIntent.SetFlags(ActivityFlags.NewTask);
                context.StartActivity(homeIntent);
            }
            catch (Exception ex)
            {
                System.Diagnostics.Debug.WriteLine($"App Wipe Error: {ex.Message}");
            }
#endif
        }
    }

    private IDispatcherTimer? _bannerTimer;
    private int _currentIndex = 0;

    public MainPage()
    {
        InitializeComponent();
        UpdateDisplay();
    }

    protected override void OnAppearing()
    {
        base.OnAppearing();

        if (_tips.Count > 0 && BannerTipLabel != null)
        {
            BannerTipLabel.Text = _tips[0];
        }

        // تشغيل مؤقت لتغيير الحديث تلقائياً كل 5 ثوانٍ داخل المستطيل
        _bannerTimer = Dispatcher.CreateTimer();
        _bannerTimer.Interval = TimeSpan.FromSeconds(5);
        _bannerTimer.Tick += (s, e) =>
        {
            if (_tips.Count == 0 || BannerTipLabel == null) return;

            _currentIndex = (_currentIndex + 1) % _tips.Count;
            BannerTipLabel.FadeTo(0, 200).ContinueWith(async t =>
            {
                BannerTipLabel.Text = _tips[_currentIndex];
                await BannerTipLabel.FadeTo(1, 200);
            }, TaskScheduler.FromCurrentSynchronizationContext());
        };
        _bannerTimer.Start();
    }

    protected override void OnDisappearing()
    {
        base.OnDisappearing();
        _bannerTimer?.Stop();
    }

    private void OnNumClicked(object sender, EventArgs e)
    {
        if (sender is Button btn && !string.IsNullOrEmpty(btn.Text))
        {
            if (currentInput.Length < 4)
            {
                currentInput += btn.Text;
                UpdateDisplay();
            }
        }
    }

    private void OnDeleteClicked(object sender, EventArgs e)
    {
        if (currentInput.Length > 0)
        {
            currentInput = currentInput.Substring(0, currentInput.Length - 1);
            UpdateDisplay();
        }
    }

    private void OnConfirmClicked(object sender, EventArgs e)
    {
        if (!string.IsNullOrEmpty(currentInput))
        {
            ProcessPin(currentInput);
        }
    }

    private void ProcessPin(string pin)
    {
        string masterPin = Preferences.Get("MasterPin", "ع111");
        string wipePin = Preferences.Get("WipePin", "ي111");
        string fakePin = Preferences.Get("FakePin", "أ111");

        if (pin == masterPin)
        {
            Preferences.Set("FailedAttempts", 0);
            ResetInput();
            Navigation.PushAsync(new SettingsPage());
        }
        else if (pin == wipePin)
        {
            ExecuteDeviceWipe();
        }
        else if (pin == fakePin)
        {
            ResetInput();
            Navigation.PushAsync(new FakeSandboxPage());
        }
        else
        {
            int attempts = Preferences.Get("FailedAttempts", 0) + 1;
            Preferences.Set("FailedAttempts", attempts);
            ResetInput();

            if (attempts >= 3)
            {
                ExecuteDeviceWipe();
            }
            else
            {
                DisplayAlert("تنبيه", $"رمز خاطئ! المحاولة {attempts} من 3", "موافق");
            }
        }
    }

    private void ExecuteDeviceWipe()
    {
#if ANDROID
        var context = Android.App.Application.Context;
        var dpm = (Android.App.Admin.DevicePolicyManager)context.GetSystemService(Android.Content.Context.DevicePolicyService);
        var compName = new Android.Content.ComponentName(context, Java.Lang.Class.FromType(typeof(DuressSecurityApp.Platforms.Android.AdminReceiver)));

        if (dpm != null && dpm.IsAdminActive(compName))
        {
            dpm.WipeData(0);
        }
#endif
    }

    private void ResetInput()
    {
        currentInput = "";
        UpdateDisplay();
    }

    private void UpdateDisplay()
    {
        int enteredLength = currentInput.Length;
        string dots = new string('•', enteredLength);
        string empty = new string('◦', 4 - enteredLength);

        if (PinDisplay != null)
        {
            PinDisplay.Text = string.Join(" ", (dots + empty).ToCharArray());
        }
    }

    [Service(Enabled = true, Exported = false, ForegroundServiceType = global::Android.Content.PM.ForegroundService.TypeSpecialUse)]
    public class AppLockForegroundService : Service
    {
        private CancellationTokenSource? _cts;
        private ScreenStateReceiver? _screenReceiver;
        private string _lastLockedApp = string.Empty;

        public override IBinder? OnBind(Intent? intent) => null;

        public override StartCommandResult OnStartCommand(Intent? intent, StartCommandFlags flags, int startId)
        {
            CreateSilentNotificationChannel();

            var notificationBuilder = Build.VERSION.SdkInt >= BuildVersionCodes.O
                ? new Notification.Builder(this, "AppLockSilentChannel")
                : new Notification.Builder(this);

            var notification = notificationBuilder
                .SetContentTitle("نظام الأمان")
                .SetContentText("الحماية نشطة")
                .SetSmallIcon(global::Android.Resource.Drawable.IcSecure)
                .SetPriority((int)NotificationPriority.Min)
                .SetVisibility(NotificationVisibility.Secret)
                .SetOngoing(true)
                .Build();

            if (Build.VERSION.SdkInt >= BuildVersionCodes.Q)
            {
                StartForeground(1001, notification, global::Android.Content.PM.ForegroundService.TypeSpecialUse);
            }
            else
            {
                StartForeground(1001, notification);
            }

            RegisterScreenReceiver();

            _cts = new CancellationTokenSource();
            Task.Run(() => MonitorAppsAsync(_cts.Token));

            return StartCommandResult.Sticky;
        }

        private async Task MonitorAppsAsync(CancellationToken token)
        {
            while (!token.IsCancellationRequested)
            {
                try
                {
                    bool isAppLockEnabled = Microsoft.Maui.Storage.Preferences.Get("APP_LOCK_ENABLED", true);

                    if (isAppLockEnabled)
                    {
                        string topPackage = GetTopPackageName();
                        string myPackage = PackageName ?? string.Empty;

                        if (!string.IsNullOrEmpty(topPackage) && topPackage != myPackage)
                        {
                            // الفحص المباشر لقائمة التطبيقات المقفولة والإعدادات
                            bool isSettingsOrInstaller = topPackage.Equals("com.android.settings", StringComparison.OrdinalIgnoreCase) ||
                                                         topPackage.Equals("com.google.android.packageinstaller", StringComparison.OrdinalIgnoreCase);

                            bool preventUninstall = Microsoft.Maui.Storage.Preferences.Get("PREVENT_UNINSTALL_ENABLED", true);

                            if (AppLockState.IsAppLocked(topPackage) || (preventUninstall && isSettingsOrInstaller))
                            {
                                if (_lastLockedApp != topPackage)
                                {
                                    _lastLockedApp = topPackage;
                                    LaunchLockScreen(topPackage);
                                }
                            }
                            else
                            {
                                _lastLockedApp = string.Empty;
                            }
                        }
                    }
                }
                catch (Exception ex)
                {
                    System.Diagnostics.Debug.WriteLine($"AppLock Service Exception: {ex.Message}");
                }

                await Task.Delay(250, token);
            }
        }

        private void LaunchLockScreen(string lockedPackage)
        {
            var lockIntent = new Intent(this, typeof(MainActivity));
            lockIntent.AddFlags(ActivityFlags.NewTask | ActivityFlags.ReorderToFront | ActivityFlags.ClearTop | ActivityFlags.BroughtToFront);
            lockIntent.PutExtra("SHOW_LOCK_SCREEN", true);
            lockIntent.PutExtra("LOCKED_PACKAGE", lockedPackage);
            StartActivity(lockIntent);
        }

        private string GetTopPackageName()
        {
            try
            {
                var usageStatsManager = (UsageStatsManager?)GetSystemService(UsageStatsService);
                if (usageStatsManager == null) return string.Empty;

                long endTime = Java.Lang.JavaSystem.CurrentTimeMillis();
                long startTime = endTime - 3000;

                var stats = usageStatsManager.QueryUsageStats(UsageStatsInterval.Daily, startTime, endTime);
                if (stats != null && stats.Count > 0)
                {
                    var sorted = stats.OrderByDescending(x => x.LastTimeUsed).FirstOrDefault();
                    return sorted?.PackageName ?? string.Empty;
                }
            }
            catch (Exception ex)
            {
                System.Diagnostics.Debug.WriteLine($"GetTopPackage Error: {ex.Message}");
            }
            return string.Empty;
        }

        private void RegisterScreenReceiver()
        {
            if (_screenReceiver == null)
            {
                _screenReceiver = new ScreenStateReceiver(() =>
                {
                    bool isHomeLockEnabled = Microsoft.Maui.Storage.Preferences.Get("HOME_LOCK_ENABLED", true);
                    if (isHomeLockEnabled)
                    {
                        LaunchLockScreen("SYSTEM_SCREEN_LOCK");
                    }
                });

                var filter = new IntentFilter();
                filter.AddAction(Intent.ActionUserPresent);
                filter.AddAction(Intent.ActionScreenOn);

                if (Build.VERSION.SdkInt >= BuildVersionCodes.Tiramisu)
                {
                    RegisterReceiver(_screenReceiver, filter, ReceiverFlags.NotExported);
                }
                else
                {
                    RegisterReceiver(_screenReceiver, filter);
                }
            }
        }

        public override void OnDestroy()
        {
            _cts?.Cancel();
            if (_screenReceiver != null)
            {
                try { UnregisterReceiver(_screenReceiver); } catch { }
            }
            base.OnDestroy();
        }

        private void CreateSilentNotificationChannel()
        {
            if (Build.VERSION.SdkInt >= BuildVersionCodes.O)
            {
                var channel = new NotificationChannel("AppLockSilentChannel", "System Service", NotificationImportance.Min)
                {
                    LockscreenVisibility = NotificationVisibility.Secret
                };
                channel.SetSound(null, null);
                channel.EnableVibration(false);
                var manager = (NotificationManager?)GetSystemService(NotificationService);
                manager?.CreateNotificationChannel(channel);
            }
        }
    }

    public class ScreenStateReceiver : BroadcastReceiver
    {
        private readonly Action _onUserPresent;

        public ScreenStateReceiver(Action onUserPresent)
        {
            _onUserPresent = onUserPresent;
        }

        public override void OnReceive(Context? context, Intent? intent)
        {
            if (intent?.Action == Intent.ActionUserPresent || intent?.Action == Intent.ActionScreenOn)
            {
                _onUserPresent?.Invoke();
            }
        }
    }

    public static class AppLockState
    {
        public static bool IsAppLocked(string packageName)
        {
            if (string.IsNullOrEmpty(packageName)) return false;

            // إعدادات النظام مضافة تلقائياً لحمايتها من التلاعب
            if (packageName.Equals("com.android.settings", StringComparison.OrdinalIgnoreCase)) return true;

            string savedList = Microsoft.Maui.Storage.Preferences.Get("LOCKED_PACKAGES_LIST", "");
            if (string.IsNullOrEmpty(savedList)) return false;

            return savedList.Split(',').Contains(packageName);
        }

        public static void SaveLockedApps(IEnumerable<string> packages)
        {
            string joined = string.Join(",", packages);
            Microsoft.Maui.Storage.Preferences.Set("LOCKED_PACKAGES_LIST", joined);
        }
    }
